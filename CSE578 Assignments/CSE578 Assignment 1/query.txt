SELECT a.Name, a.AttractionID, a.Category, a.Type, 
	AVG(CAST(substr(c.duration, 1, 2) AS INTEGER) * 3600 + CAST(substr(c.duration, 4, 2) AS INTEGER) * 60 + CAST(substr(c.duration, 7, 2) AS INTEGER)) AS average_duration_in_seconds,
	count(1) Cnt
	FROM checkin c 
		JOIN attraction a 
		ON c.attraction = a.AttractionID 
	WHERE (a.Category LIKE '%Ride%' OR a.Type LIKE '%Ride%') 
	AND c.duration IS NOT NULL 
	AND CAST(substr(c.duration, 1, 2) AS INTEGER) * 3600 + CAST(substr(c.duration, 4, 2) AS INTEGER) * 60 + CAST(substr(c.duration, 7, 2) AS INTEGER) > 0
	AND c.duration GLOB '[0-9]:[0-5][0-9]:[0-5][0-9]' 
GROUP BY a.Name, a.Category, a.AttractionID 
ORDER BY average_duration_in_seconds DESC
LIMIT 2;
